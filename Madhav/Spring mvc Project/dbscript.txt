
Admin Table

CREATE TABLE  "ADMIN" 
   (	"ADMINID" NUMBER, 
	"MAILID" VARCHAR2(30), 
	"PASSWORD" VARCHAR2(20), 
	"NAME" VARCHAR2(20), 
	 PRIMARY KEY ("ADMINID") ENABLE, 
	 UNIQUE ("MAILID") ENABLE
   ) ;

   (	"STUD_ID" NUMBER,
